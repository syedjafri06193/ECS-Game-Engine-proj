cmake_minimum_required(VERSION 3.25)
project(ECSGameEngine LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(ENGINE_BUILD_TESTS   "Build unit tests"                           ON)
option(ENGINE_BUILD_BENCH   "Build benchmarks"                           ON)
option(ENGINE_BUILD_VULKAN  "Build the Vulkan renderer and demo app"     OFF)
option(ENGINE_TRACY         "Enable Tracy profiler zones"                OFF)
option(ENGINE_WARNINGS_AS_ERRORS "Treat warnings as errors"              ON)
set(ENGINE_SANITIZE "" CACHE STRING "Sanitizer: address, thread, undefined, or empty")

include(cmake/Dependencies.cmake)
include(cmake/Warnings.cmake)

if(ENGINE_SANITIZE STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(ENGINE_SANITIZE STREQUAL "thread")
  add_compile_options(-fsanitize=thread)
  add_link_options(-fsanitize=thread)
elseif(ENGINE_SANITIZE STREQUAL "undefined")
  add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all)
  add_link_options(-fsanitize=undefined)
endif()

add_subdirectory(engine)
add_subdirectory(gameplay)

if(ENGINE_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(ENGINE_BUILD_BENCH)
  add_subdirectory(apps/bench)
endif()
if(ENGINE_BUILD_VULKAN)
  add_subdirectory(apps/demo)
endif()
